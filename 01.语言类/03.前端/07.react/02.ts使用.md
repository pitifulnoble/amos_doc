# TS

## TS类型

```typescript
// 基础类型
const name: string = 'John';
const age: number = 20;
const isStudent: boolean = true;
let a: undefined;
const b: null = null;
const user: object = {};
const big: bigint = 100n;
const sym: symbol = Symbol('amos');

// 数组类型
const list1:number[] = [1, 2, 3];
const list2:Array<number> = [1, 2, 3];
const list3:[number ,string, boolean] = [1, '4', true];
const list4:[{name: string, age: number}] = [{name: 'John', age: 20}];

interface Person {
  name: string;
  age: number;
}
const list5:Person[] = [{name: 'John', age: 20}];

//函数类型
function add(a: number, b: number): number {
  return a + b;
}
function add2(a: number, b: number): void {}
const add5:(a: number, b: number) => number = (a, b) => a + b;
```
